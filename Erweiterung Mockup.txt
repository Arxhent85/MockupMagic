<script>
    const container = document.getElementById('canvas-container');
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(40, container.clientWidth / container.clientHeight, 0.1, 100);
    camera.position.set(0, 0, 35);

    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true, preserveDrawingBuffer: true });
    renderer.setSize(container.clientWidth, container.clientHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); 
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.outputEncoding = THREE.sRGBEncoding;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    container.appendChild(renderer.domElement);

    const controls = new THREE.OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.05;
    controls.target.set(0, 0, 0); 

    const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
    scene.add(ambientLight);
    const mainLight = new THREE.DirectionalLight(0xffffff, 1.2);
    mainLight.position.set(10, 20, 15);
    mainLight.castShadow = true;
    scene.add(mainLight);
    const fillLight = new THREE.DirectionalLight(0xe2e8f0, 0.8);
    fillLight.position.set(-15, 5, -10);
    scene.add(fillLight);

    // Globale Materialien
    const bodyMat = new THREE.MeshStandardMaterial({ color: 0xd1d5db, metalness: 1.0, roughness: 0.25 });
    const lidMat = new THREE.MeshStandardMaterial({ color: 0x111111, metalness: 0.1, roughness: 0.2 }); 

    const uvCanvas = document.createElement('canvas');
    uvCanvas.width = 4096; uvCanvas.height = 4096;
    const uvCtx = uvCanvas.getContext('2d');
    uvCtx.fillStyle = 'rgba(255, 255, 255, 0)';
    uvCtx.clearRect(0, 0, uvCanvas.width, uvCanvas.height);

    const canvasTexture = new THREE.CanvasTexture(uvCanvas);
    canvasTexture.encoding = THREE.sRGBEncoding;
    canvasTexture.flipY = false;
    
    const labelMat = new THREE.MeshStandardMaterial({
        map: canvasTexture,
        transparent: true,
        alphaTest: 0.05, 
        metalness: 0.1,
        roughness: 0.3,
        side: THREE.FrontSide 
    });

    // --- 3D MODELLE AUFBAUEN ---
    let currentModelType = 'can'; 

    const masterGroup = new THREE.Group(); 
    scene.add(masterGroup);
    
    // Gruppe 1: Sprühdose (Bestehend)
    const canGroup = new THREE.Group(); 
    masterGroup.add(canGroup);
    const geoCylinder = new THREE.CylinderGeometry(1, 1, 1, 64);
    const geoTorus = new THREE.TorusGeometry(1, 0.05, 16, 64);
    const geoDome = new THREE.SphereGeometry(1, 64, 32, 0, Math.PI*2, 0, Math.PI/2); 

    const meshBody = new THREE.Mesh(geoCylinder, bodyMat);
    const meshLabel = new THREE.Mesh(geoCylinder, labelMat);
    meshLabel.visible = false; 
    const meshRimBottom = new THREE.Mesh(geoTorus, bodyMat);
    meshRimBottom.rotation.x = Math.PI / 2;
    const meshRimTop = new THREE.Mesh(geoTorus, bodyMat);
    meshRimTop.rotation.x = Math.PI / 2;
    const meshDome = new THREE.Mesh(geoDome, bodyMat);
    meshDome.scale.y = 0.4;
    const meshValve = new THREE.Mesh(new THREE.CylinderGeometry(0.2, 0.2, 0.4, 32), bodyMat);
    const meshLid = new THREE.Mesh(new THREE.CylinderGeometry(1, 1, 1, 64), lidMat);
    canGroup.add(meshBody, meshLabel, meshRimBottom, meshRimTop, meshDome, meshValve, meshLid);

    // --- NEU: Gruppe 2: Silikonkartusche (Exakt nach Foto) ---
    const cartGroup = new THREE.Group();
    cartGroup.visible = false; 
    masterGroup.add(cartGroup);
    
    const meshCartBody = new THREE.Mesh(geoCylinder, bodyMat);
    const meshCartLabel = new THREE.Mesh(geoCylinder, labelMat);
    meshCartLabel.visible = false;
    
    // Oberer Rand (Dicke Plastiklippe)
    const meshCartTopLip = new THREE.Mesh(new THREE.CylinderGeometry(1, 1, 1, 64), bodyMat);
    
    // Flacher Deckel (innerhalb der Lippe)
    const meshCartTopFlat = new THREE.Mesh(new THREE.CylinderGeometry(0.9, 0.9, 1, 64), bodyMat);
    
    // Gewinde-Basis (Zylinder mit Rillen angedeutet durch flatShading oder einfach glatt)
    const meshCartThread = new THREE.Mesh(new THREE.CylinderGeometry(0.25, 0.25, 1, 32), bodyMat);
    
    // Düsen-Basis (Der dicke Teil der aufgeschraubt wird)
    const meshCartNozzleBase = new THREE.Mesh(new THREE.CylinderGeometry(0.2, 0.32, 1, 32), lidMat);
    
    // Düsen-Spitze (Der lange Konus)
    const meshCartNozzleTip = new THREE.Mesh(new THREE.CylinderGeometry(0.04, 0.2, 1, 32), lidMat);
    
    // Unterer Kolbenrand
    const meshCartRim = new THREE.Mesh(new THREE.CylinderGeometry(1.02, 1.02, 0.1, 64), bodyMat);
    
    cartGroup.add(meshCartBody, meshCartLabel, meshCartTopLip, meshCartTopFlat, meshCartThread, meshCartNozzleBase, meshCartNozzleTip, meshCartRim);


    // --- DYNAMISCHE MAßE & POSITIONIERUNG ---
    function updateDimensions() {
        const d_mm = parseFloat(document.getElementById('dim-diam').value);
        const h_mm = parseFloat(document.getElementById('dim-height').value);
        const l_mm = parseFloat(document.getElementById('dim-lidheight').value);

        const r = (d_mm / 2) / 10; 
        const h = h_mm / 10;
        const l = l_mm / 10;

        if (currentModelType === 'can') {
            canGroup.visible = true;
            cartGroup.visible = false;

            meshBody.scale.set(r, h, r); meshBody.position.y = h / 2;
            meshLabel.scale.set(r * 1.005, h, r * 1.005); meshLabel.position.y = h / 2;
            meshRimBottom.scale.set(r, r, r); meshRimBottom.position.y = 0;
            meshRimTop.scale.set(r, r, r); meshRimTop.position.y = h;
            meshDome.scale.set(r * 0.95, r * 0.4, r * 0.95); meshDome.position.y = h;
            meshValve.position.y = h + (r*0.4);
            meshLid.scale.set(r, l, r); meshLid.position.y = h + (l / 2);

            const showLid = document.getElementById('toggle-lid').checked;
            meshLid.visible = showLid;

            let totalHeight = h;
            if(showLid) totalHeight += l;
            canGroup.position.y = - (totalHeight / 2);

        } else if (currentModelType === 'cartridge') {
            canGroup.visible = false;
            cartGroup.visible = true;

            // Hauptkörper
            meshCartBody.scale.set(r, h, r); meshCartBody.position.y = h / 2;
            meshCartLabel.scale.set(r * 1.005, h, r * 1.005); meshCartLabel.position.y = h / 2;
            meshCartRim.scale.set(r, 1, r); meshCartRim.position.y = 0.05;

            // Kartuschen-Kopf (Flach mit Lippe)
            const lipH = 0.15; // 1.5mm Lippe oben
            meshCartTopLip.scale.set(r, lipH, r);
            meshCartTopLip.position.y = h + (lipH / 2);

            const flatH = 0.05; // Deckeldicke
            meshCartTopFlat.scale.set(r, flatH, r);
            meshCartTopFlat.position.y = h + lipH - (flatH / 2); // Leicht nach innen versetzt

            // Gewinde (Standard ca. 1 cm hoch)
            const threadH = 1.0; 
            meshCartThread.scale.set(r, threadH, r);
            meshCartThread.position.y = h + lipH + (threadH / 2);

            // Düse (l_mm = Gesamtlänge der Düse). Aufgeteilt in Basis und Spitze.
            const nozzleBaseH = l * 0.15; // Untere 15% sind der dicke Schraubverschluss
            const nozzleTipH = l * 0.85;  // Obere 85% sind die dünne Spitze

            meshCartNozzleBase.scale.set(r, nozzleBaseH, r);
            meshCartNozzleBase.position.y = h + lipH + (nozzleBaseH / 2); // Schraubt sich über das Gewinde

            meshCartNozzleTip.scale.set(r, nozzleTipH, r);
            meshCartNozzleTip.position.y = h + lipH + nozzleBaseH + (nozzleTipH / 2);

            // Toggle Logik (Nur die Düse wird ausgeblendet, Gewinde bleibt)
            const showNozzle = document.getElementById('toggle-lid').checked;
            meshCartNozzleBase.visible = showNozzle;
            meshCartNozzleTip.visible = showNozzle;

            // Zentrierung
            let totalHeight = h + lipH + threadH; // Höhe ohne Düse
            if(showNozzle) totalHeight = h + lipH + nozzleBaseH + nozzleTipH;
            cartGroup.position.y = - (totalHeight / 2);
        }

        if(uploadedImage) updateTexture();
    }

    // --- ETIKETTEN LOGIK ---
    let uploadedImage = null;
    function updateTexture() {
        if (!uploadedImage) return;
        const wrapPercent = parseFloat(document.getElementById('decal-scale').value);
        const rotX = parseFloat(document.getElementById('decal-x').value);
        const posY = parseFloat(document.getElementById('decal-y').value);
        const d_mm = parseFloat(document.getElementById('dim-diam').value);
        const h_mm = parseFloat(document.getElementById('dim-height').value);
        
        const physCircumference = Math.PI * d_mm;
        const physHeight = h_mm;
        const cw = uvCanvas.width;
        const ch = uvCanvas.height;
        uvCtx.clearRect(0, 0, cw, ch);

        const imgAspect = uploadedImage.width / uploadedImage.height;
        const targetPhysWidth = physCircumference * wrapPercent;
        const targetPhysHeight = targetPhysWidth / imgAspect; 

        const drawW = (targetPhysWidth / physCircumference) * cw;
        const drawH = (targetPhysHeight / physHeight) * ch;
        const centerX = rotX * cw;
        const centerY = posY * ch; 
        const startX = centerX - (drawW / 2);
        const startY = centerY - (drawH / 2);

        uvCtx.save();
        uvCtx.translate(0, ch);
        uvCtx.scale(1, -1);
        uvCtx.drawImage(uploadedImage, startX, startY, drawW, drawH);
        if (startX < 0) uvCtx.drawImage(uploadedImage, startX + cw, startY, drawW, drawH);
        if (startX + drawW > cw) uvCtx.drawImage(uploadedImage, startX - cw, startY, drawW, drawH);
        uvCtx.restore();
        canvasTexture.needsUpdate = true;
    }

    document.getElementById('file-upload').addEventListener('change', async (e) => {
        const file = e.target.files[0];
        if (!file) return;
        document.getElementById('upload-overlay').classList.add('opacity-100');
        try {
            if (file.type === 'application/pdf') {
                const fileReader = new FileReader();
                fileReader.onload = async function() {
                    const typedarray = new Uint8Array(this.result);
                    const pdf = await pdfjsLib.getDocument(typedarray).promise;
                    const page = await pdf.getPage(1);
                    const viewport = page.getViewport({ scale: 4.0 }); 
                    const tempCanvas = document.createElement('canvas');
                    const tempCtx = tempCanvas.getContext('2d');
                    tempCanvas.width = viewport.width;
                    tempCanvas.height = viewport.height;
                    await page.render({ canvasContext: tempCtx, viewport: viewport }).promise;
                    const img = new Image();
                    img.onload = () => enableDecalControls(img, "PDF in HD geladen!");
                    img.src = tempCanvas.toDataURL('image/png');
                };
                fileReader.readAsArrayBuffer(file);
            } else {
                const url = URL.createObjectURL(file);
                const img = new Image();
                img.onload = () => enableDecalControls(img, "Bild geladen!");
                img.src = url;
            }
        } catch (error) {
            document.getElementById('upload-overlay').classList.remove('opacity-100');
            document.getElementById('upload-text').innerText = "Fehler. Erneut versuchen.";
        }
    });

    document.getElementById('btn-export-glb').addEventListener('click', async () => {
        const overlay = document.getElementById('export-overlay');
        const modal = document.getElementById('download-ready-modal');
        const finalLink = document.getElementById('final-download-link');
        
        overlay.classList.remove('opacity-0', 'pointer-events-none');
        await new Promise(resolve => setTimeout(resolve, 100));

        try {
            const exporter = new THREE.GLTFExporter();
            
            // Wichtig: Wir exportieren nur die Gruppe, die gerade sichtbar ist!
            const groupToExport = currentModelType === 'can' ? canGroup : cartGroup;

            exporter.parse(groupToExport, async function ( gltf ) {
                
                const isGLB = gltf instanceof ArrayBuffer;
                const blob = isGLB ? 
                    new Blob([gltf], { type: 'model/gltf-binary' }) : 
                    new Blob([JSON.stringify(gltf)], { type: 'text/plain' });
                    
                const filename = currentModelType === 'can' ? "Spruehdose.glb" : "Kartusche.glb";
                const file = new File([blob], filename, { type: blob.type });

                // Mobile Share Fallback
                if (navigator.canShare && navigator.canShare({ files: [file] })) {
                    try {
                        overlay.classList.add('opacity-0', 'pointer-events-none');
                        await navigator.share({ files: [file], title: '3D Modell' });
                        return; 
                    } catch (err) {}
                }

                // PC Object URL
                const objectUrl = URL.createObjectURL(blob);
                const tempLink = document.createElement('a');
                tempLink.style.display = 'none';
                tempLink.href = objectUrl;
                tempLink.download = filename;
                document.body.appendChild(tempLink);
                tempLink.click();
                document.body.removeChild(tempLink);
                
                finalLink.href = objectUrl;
                finalLink.download = filename;
                
                overlay.classList.add('opacity-0', 'pointer-events-none');
                modal.classList.remove('opacity-0', 'pointer-events-none');

            }, { binary: true });

        } catch(e) {
            console.error(e);
            alert("Fehler beim Erstellen der 3D-Datei.");
            overlay.classList.add('opacity-0', 'pointer-events-none');
        }
    });
